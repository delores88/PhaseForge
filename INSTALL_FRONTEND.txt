# PHASEFORGE FRONTEND - WINDOWS INSTALL
# Copy this entire file, open PowerShell in the extracted PhaseForge project root,
# paste it, and press Enter. Every explanatory line is a PowerShell comment.

$ErrorActionPreference = "Stop"
Set-StrictMode -Version Latest

$PFRepoRoot = (Get-Location).Path
$PFFrontendRoot = Join-Path $PFRepoRoot "frontend"
$PFMinimumNode = [version]"20.9.0"

function PF-Step([string]$Message) {
    Write-Host "`n==> $Message" -ForegroundColor Cyan
}

function PF-RefreshPath {
    $paths = @(
        [Environment]::GetEnvironmentVariable("Path", "Process"),
        [Environment]::GetEnvironmentVariable("Path", "User"),
        [Environment]::GetEnvironmentVariable("Path", "Machine")
    ) | Where-Object { -not [string]::IsNullOrWhiteSpace($_) }
    $env:Path = ($paths -join ";")
}

function PF-NodeVersion {
    if ($null -eq (Get-Command "node" -ErrorAction SilentlyContinue)) {
        return $null
    }
    $value = (& node -p "process.versions.node").Trim()
    if ($LASTEXITCODE -ne 0 -or [string]::IsNullOrWhiteSpace($value)) {
        return $null
    }
    try { return [version]$value } catch { return $null }
}

function PF-NpmCommand {
    $command = Get-Command "npm.cmd" -ErrorAction SilentlyContinue
    if ($null -eq $command) {
        $command = Get-Command "npm" -ErrorAction SilentlyContinue
    }
    if ($null -eq $command) { return $null }
    return $command.Source
}

function PF-Run([string]$Label, [string]$Program, [string[]]$Arguments) {
    PF-Step $Label
    & $Program @Arguments
    if ($LASTEXITCODE -ne 0) {
        throw "$Label failed with exit code $LASTEXITCODE."
    }
}

function PF-InstallNode([bool]$Upgrade) {
    if ($null -eq (Get-Command "winget" -ErrorAction SilentlyContinue)) {
        throw "Node.js $PFMinimumNode or newer is required and winget is unavailable. Install a current Node.js LTS release, reopen PowerShell, and paste this file again."
    }
    $verb = if ($Upgrade) { "upgrade" } else { "install" }
    PF-Run "Installing or upgrading Node.js LTS" "winget" @(
        $verb, "--id", "OpenJS.NodeJS.LTS", "--exact",
        "--accept-package-agreements", "--accept-source-agreements",
        "--disable-interactivity"
    )
    PF-RefreshPath
}

if (-not (Test-Path (Join-Path $PFFrontendRoot "package.json"))) {
    throw "frontend/package.json was not found. Open PowerShell in the extracted PhaseForge project root before pasting this file."
}

Write-Host "PhaseForge frontend installer (v0.8.0-alpha.1)" -ForegroundColor White
Write-Host "Repository: $PFRepoRoot"
Write-Host "Frontend:   $PFFrontendRoot"
Write-Host "Node minimum: $PFMinimumNode"

PF-RefreshPath
$nodeVersion = PF-NodeVersion
if ($null -eq $nodeVersion) {
    PF-InstallNode $false
    $nodeVersion = PF-NodeVersion
} elseif ($nodeVersion -lt $PFMinimumNode) {
    PF-InstallNode $true
    $nodeVersion = PF-NodeVersion
}
if ($null -eq $nodeVersion -or $nodeVersion -lt $PFMinimumNode) {
    throw "A supported Node.js installation was not found after installation. Close and reopen PowerShell in this project root, then paste INSTALL_FRONTEND.txt again."
}

$npm = PF-NpmCommand
if ([string]::IsNullOrWhiteSpace($npm)) {
    throw "npm was not found beside Node.js. Repair the Node.js LTS installation and paste this file again."
}

PF-Step "Reporting frontend tool versions"
Write-Host "Node.js v$nodeVersion accepted (minimum $PFMinimumNode)." -ForegroundColor Green
& $npm --version
if ($LASTEXITCODE -ne 0) { throw "npm could not start." }

$env:NEXT_TELEMETRY_DISABLED = "1"
Push-Location $PFFrontendRoot
try {
    if (-not (Test-Path ".env.local")) {
        if (-not (Test-Path ".env.example")) {
            throw "frontend/.env.example is missing."
        }
        Copy-Item ".env.example" ".env.local"
        Write-Host "Created frontend/.env.local from frontend/.env.example."
    }

    if (Test-Path "package-lock.json") {
        PF-Run "Installing the locked frontend dependencies" $npm @(
            "ci", "--no-audit", "--no-fund"
        )
    } else {
        PF-Run "Installing frontend dependencies and creating package-lock.json" $npm @(
            "install", "--no-audit", "--no-fund"
        )
    }

    # The production build is the authoritative dependency and integration check.
    PF-Run "Building the production browser client" $npm @("run", "build")

    $exportedIndex = Join-Path $PFFrontendRoot "out\index.html"
    if (-not (Test-Path $exportedIndex)) {
        throw "The frontend build completed without producing out/index.html."
    }
} finally {
    Pop-Location
}

PF-Step "Frontend installation complete"
Write-Host "Static build: frontend\out\index.html" -ForegroundColor Green
Write-Host "Next: use START_FRONTEND.txt or START_ALL.txt." -ForegroundColor Green
